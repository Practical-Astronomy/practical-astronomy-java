90 zoneCorrection, localDay, localMonth, localYear);
91 int gMonth = Macros.localCivilTimeGreenwichMonth(lctHours, lctMinutes, lctSeconds, daylightSaving,
92 zoneCorrection, localDay, localMonth, localYear);
93 int gYear = Macros.localCivilTimeGreenwichYear(lctHours, lctMinutes, lctSeconds, daylightSaving, zoneCorrection,
94 localDay, localMonth, localYear);
95 double trueAnomalyDeg = Macros.sunTrueAnomaly(lctHours, lctMinutes, lctSeconds, daylightSaving, zoneCorrection,
96 localDay, localMonth, localYear);
97 double trueAnomalyRad = Math.toRadians(trueAnomalyDeg);
98 double eccentricity = Macros.sunEcc(gDay, gMonth, gYear);
99 double f = (1 + eccentricity * Math.cos(trueAnomalyRad)) / (1 - eccentricity * eccentricity);
100 double rKm = 149598500 / f;
101 double thetaDeg = f * 0.533128;
102
103 double sunDistKm = Util.round(rKm, 0);
104 double sunAngSizeDeg = Macros.decimalDegreesDegrees(thetaDeg);
105 double sunAngSizeMin = Macros.decimalDegreesMinutes(thetaDeg);
106 double sunAngSizeSec = Macros.decimalDegreesSeconds(thetaDeg);
