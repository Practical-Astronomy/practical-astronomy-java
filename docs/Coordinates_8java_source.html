524 public HeliographicCoordinates heliographicCoordinates(double helioPositionAngleDeg, double helioDisplacementArcmin,
525 double gwdateDay, int gwdateMonth, int gwdateYear) {
526 double julianDateDays = Macros.civilDateToJulianDate(gwdateDay, gwdateMonth, gwdateYear);
527 double tCenturies = (julianDateDays - 2415020) / 36525;
528 double longAscNodeDeg = Macros.degreesMinutesSecondsToDecimalDegrees(74, 22, 0) + (84 * tCenturies / 60);
529 double sunLongDeg = Macros.sunLong(0, 0, 0, 0, 0, gwdateDay, gwdateMonth, gwdateYear);
530 double y = Math.sin(Math.toRadians(longAscNodeDeg - sunLongDeg))
531 * Math.cos(Math.toRadians(Macros.degreesMinutesSecondsToDecimalDegrees(7, 15, 0)));
532 double x = -Math.cos(Math.toRadians(longAscNodeDeg - sunLongDeg));
533 double aDeg = Macros.wToDegrees(Math.atan2(y, x));
534 double mDeg1 = 360 - (360 * (julianDateDays - 2398220) / 25.38);
535 double mDeg2 = mDeg1 - 360 * Math.floor(mDeg1 / 360);
536 double l0Deg1 = mDeg2 + aDeg;
537 double b0Rad = Math.asin(Math.sin(Math.toRadians(sunLongDeg - longAscNodeDeg))
538 * Math.sin(Math.toRadians(Macros.degreesMinutesSecondsToDecimalDegrees(7, 15, 0))));
539 double theta1Rad = Math.atan(-Math.cos(Math.toRadians(sunLongDeg))
540 * Math.tan(Math.toRadians(Macros.obliq(gwdateDay, gwdateMonth, gwdateYear))));
541 double theta2Rad = Math.atan(-Math.cos(Math.toRadians(longAscNodeDeg - sunLongDeg))
542 * Math.tan(Math.toRadians(Macros.degreesMinutesSecondsToDecimalDegrees(7, 15, 0))));
543 double pDeg = Macros.wToDegrees(theta1Rad + theta2Rad);
544 double rho1Deg = helioDisplacementArcmin / 60;
545 double rhoRad = Math.asin(2 * rho1Deg / Macros.sunDia(0, 0, 0, 0, 0, gwdateDay, gwdateMonth, gwdateYear))
546 - Math.toRadians(rho1Deg);
547 double bRad = Math.asin(Math.sin(b0Rad) * Math.cos(rhoRad)
548 + Math.cos(b0Rad) * Math.sin(rhoRad) * Math.cos(Math.toRadians(pDeg - helioPositionAngleDeg)));
549 double bDeg = Macros.wToDegrees(bRad);
550 double lDeg1 = Macros.wToDegrees(
551 Math.asin(Math.sin(rhoRad) * Math.sin(Math.toRadians(pDeg - helioPositionAngleDeg)) / Math.cos(bRad)))
552 + l0Deg1;
553 double lDeg2 = lDeg1 - 360 * Math.floor(lDeg1 / 360);
554
555 double helioLongDeg = Util.round(lDeg2, 2);
556 double helioLatDeg = Util.round(bDeg, 2);
557
558 return new HeliographicCoordinates(helioLongDeg, helioLatDeg);
